_PG = {}


def _page(i):
    key = ("pg", i)
    if key in _PG:
        return _PG[key]
    titles = [("富士山下", "吉他谱 · 完整版"), ("富士山下", "吉他谱 · 完整版"), ("富士山下", "吉他谱 · 完整版")]
    starts = [0, 9, 18]
    pg = pages.build_page(list(range(starts[i], starts[i] + 9)), titles[i][0], titles[i][1], i + 1, 3)
    _PG[key] = pg
    return pg


def _page_turn_quad(spine_x, top, w, h, f):
    """Quad for a page hinged at its left spine, horizontal compression f (1 flat, 0 edge-on)."""
    c = max(-1.0, min(1.0, f))
    s = math.sin(math.acos(abs(c))) * (1 if c >= 0 else -1)
    xr = spine_x + w * c
    dy = h * 0.06 * abs(s)
    return np.array(
        [[spine_x, top], [xr, top + dy], [xr, top + h - dy], [spine_x, top + h]], F32
    )


def _dof(img, focus, radius, strength=0.9):
    blur = cv2.GaussianBlur(img, (0, 0), 9)
    yy, xx = np.mgrid[0 : img.shape[0], 0 : img.shape[1]].astype(F32)
    d = np.sqrt((xx - focus[0]) ** 2 + (yy - focus[1]) ** 2) / radius
    m = np.clip(1 - d * d, 0, 1)[..., None] ** 0.8
    return np.clip(img * m + blur * (1 - m) * strength, 0, 1)


def scene5(t):
    img = studio_bg("warm").copy()
    dust(img, t, gain=0.5)
    # table light pool
    lamp = glow_field(D.W, D.H, (D.W * 0.44, D.H * 0.22), 900, D.hexc("#6A4218"), falloff=2.0, gain=0.75)
    over(img, lamp, 0, 0, 1.0, blend="add")

    p1 = _page(0)
    p2 = _page(1)
    scale = 0.56
    pw, ph = int(p1.w * scale), int(p1.h * scale)
    cx, cy = 1010, 566

    # entrance
    ent = remap(t, 36.15, 37.15, ease_out_cubic)
    lift = remap(t, 36.15, 37.2, ease_out_cubic)
    macro = remap(t, 40.45, 42.3, ease_in_out_cubic)
    back = remap(t, 42.35, 43.6, ease_in_out_cubic)

    # ---------------- macro zoom on the printed notation
    if macro > 0:
        z = 1 + macro * 1.15
        # focus on a notation area of the page
        fx = cx + pw * 0.18
        fy = cy - ph * 0.10
        scale2 = scale * z
        w2, h2 = int(p1.w * scale2), int(p1.h * scale2)
        pg = p2.scaled(w2, h2)
        px = fx - (fx - cx + pw / 2) * z
        py = fy - (fy - cy + ph / 2) * z
        over(img, shadow(pg, 60, 0.7, dx=-14, dy=18), px - 10, py + 20, ent * macro)
        over(img, pg, px, py, ent * macro)
        if macro > 0.25:
            img = _dof(img, (fx, fy), 520, 0.75)

    # ---------------- single page + turn
    flip = remap(t, 38.75, 40.05, ease_in_out_cubic)
    if macro <= 0.001:
        # back pages in stack
        for k, (ox, oy, rot) in enumerate([(26, -20, 2.2), (13, -10, 1.1)]):
            q = pages.page_quad(cx + ox, cy + oy, pw, ph, rz=rot, ry=-6)
            blit_perspective(img, shadow(p1, 30, 0.5), q + np.array([0, 14], F32), ent * 0.7)
            blit_perspective(img, p1, q, ent * 0.9)
        # main page / turn
        q = pages.page_quad(cx, cy, pw, ph, ry=-6)
        if flip <= 0:
            blit_perspective(img, shadow(p1, 44, 0.72, dx=-12, dy=16), q + np.array([0, 18], F32), ent)
            blit_perspective(img, p1, q, ent)
            # warm light sweep across the paper
            sw = remap(t, 37.2, 38.5, ease_in_out_cubic)
            if 0 < sw < 1:
                sy = cy - ph / 2 + sw * ph
                over(img, glow_field(D.W, 180, (D.W * 0.5, 90), 820, D.hexc("#FFD9A0"), falloff=1.8, gain=0.35), 0, sy - 90, 0.8, blend="add")
        else:
            # page turning around its left spine
            f = math.cos(math.pi * flip)
            spine = cx - pw / 2
            top = cy - ph / 2
            if f > 0.02:
                q1 = _page_turn_quad(spine, top, pw, ph, f)
                shade = 0.55 + 0.45 * abs(f)
                spr = p1
                blit_perspective(img, spr, q1, ent * min(1, abs(f) * 1.6))
                over(img, solid(int(pw * abs(f) + 2), ph, np.array([0.06, 0.04, 0.03], F32), alpha=0.30 * (1 - abs(f))), spine, top, 1.0)
            if f < 0.98:
                f2 = abs(f)
                q2 = _page_turn_quad(spine, top, pw, ph, max(0.02, f2))
                blit_perspective(img, shadow(p2, 30, 0.4), q2 + np.array([6, 10], F32), ent)
                blit_perspective(img, p2, q2, ent)
        # base shadow under the page
        over(img, shadow(p1, 50, 0.55, dx=0, dy=26), cx - pw / 2, cy - ph / 2 + 26, ent * (1 - macro))

    # ---------------- pull back to a printed stack
    if back > 0:
        sc3 = 0.5
        w3, h3 = int(p1.w * sc3), int(p1.h * sc3)
        # stack of three pages
        for k, (ox, oy, rot, spr) in enumerate([(34, 26, 3.0, _page(2)), (17, 13, 1.5, p2), (0, 0, 0.0, p1)]):
            q = pages.page_quad(cx + ox, cy + oy + 10, w3, h3, rz=rot, ry=-8)
            blit_perspective(img, shadow(spr.scaled(w3, h3), 30, 0.55), q + np.array([0, 16], F32), back)
            blit_perspective(img, spr.scaled(w3, h3), q, back)
        # top page lifts toward camera at the end
        lift2 = remap(t, 43.9, 45.5, ease_in_cubic)
        if lift2 > 0:
            q = pages.page_quad(cx, cy - lift2 * 90, w3 * (1 + lift2 * 0.5), h3 * (1 + lift2 * 0.5), rz=-2.0 * lift2, ry=-8 - 10 * lift2)
            blit_perspective(img, shadow(p1.scaled(w3, h3), 40, 0.55), q + np.array([0, 22], F32), back * (1 - lift2 * 0.6))
            blit_perspective(img, p1.scaled(w3, h3), q, back)

    # ---------------- captions
    caption(img, "A4 PDF。", 140, 320, t, 37.9, 40.4, D.PAPER, "serif", 74, 1.5, 600, rise=20, dur=0.55)
    caption(img, "可以直接打印。", 140, 424, t, 38.5, 40.4, D.hexc("#C9B9A0"), "sans", 26, tracking=2.0, rise=14, dur=0.5)
    caption(img, "300 DPI · 整页排版", 140, 900, t, 41.4, 43.6, D.hexc("#C9B9A0"), "sans", 19, tracking=2.4, rise=12, dur=0.5)
    if t > 36.4:
        p = remap(t, 36.4, 37.1, ease_out_cubic)
        lab = text_sprite("04 / 成品", D.GREY, "sans", 15, tracking=4.0)
        over(img, lab.with_opacity(0.6 * p * (1 - back * 0.8)), 140, 238)
    return img

def _logo_mark(p, line_p):
    """Viewfinder brackets + a tab line. p: 0..1 draw-on."""
    w, h = 260, 260
    spr = blank(w, h)
    B = spr.rgba[..., :3]
    line_col = D.TEAL
    # brackets
    arm = 62 * ease_out_cubic(p)
    th = 5
    pad = 26
    for (x, y, dx, dy) in [(pad, pad, 1, 1), (w - pad, pad, -1, 1), (pad, h - pad, 1, -1), (w - pad, h - pad, -1, -1)]:
        if arm < 2:
            continue
        over(B, solid(arm, th, D.PAPER), min(x, x + dx * arm), min(y, y + dy * th) - th / 2 + th / 2, 0.92)
        over(B, solid(th, arm, D.PAPER), min(x, x + dx * th) - th / 2 + th / 2, min(y, y + dy * arm), 0.92)
    # staff line across
    if line_p > 0:
        lw = int((w - 2 * pad + 40) * line_p)
        x0 = (w - lw) / 2
        over(B, solid(lw, 3, line_col), x0, h / 2 - 1.5, 1.0)
        over(B, glow(ellipse_sprite(8, 8, line_col), 8, 0.0), x0 + lw - 6, h / 2 - 4, 0.9, blend="add")
    return spr

    img = studio_bg("dark").copy()

    # ambience
    g = glow_field(D.W, D.H, (D.W * 0.5, D.H * 0.42), 820, D.hexc("#0B3D36"), falloff=2.4, gain=0.55)
    over(img, g, 0, 0, 1.0, blend="add")

    # ---------------- after-page: receding paper -> line
    if t < 47.3:
        p1 = _page(0)
        out = remap(t, 45.55, 46.9, ease_in_cubic)
        sc = 0.5 * (1 - out * 0.55)
        w, h = int(p1.w * sc), int(p1.h * sc)
        spr = p1.scaled(w, h)
        q = pages.page_quad(D.W * 0.5, D.H * 0.5 - out * 40, w, h, rz=-2 * out, ry=-8 - 12 * out)
        blit_perspective(img, shadow(spr, 50, 0.6), q + np.array([0, 18], F32), 1 - out)
        blit_perspective(img, spr, q, 1 - out)
        # the top staff line stays and glows
        lp = remap(t, 46.3, 47.4, ease_in_out_cubic)
        if lp > 0:
            ly = D.H * 0.5 - 40 * out + h * 0.06
            lw = int((D.W * 0.62) * lp)
            over(img, solid(lw, 2, D.TEAL), (D.W - lw) / 2, ly, lp * 0.9, blend="add")
            over(img, glow_field(D.W, 200, (D.W * 0.5, 100), D.W * 0.4, D.TEAL, falloff=2.0, gain=0.4), 0, ly - 100, lp * 0.8, blend="add")

    # ---------------- hero couplet

    # ---------------- logo + wordmark
    lp = remap(t, 52.85, 53.9, ease_out_cubic)
    wm = remap(t, 53.6, 54.5, ease_out_cubic)
    tg = remap(t, 54.4, 55.3, ease_out_cubic)
    if lp > 0:
        mark = _logo_mark(lp, remap(t, 53.1, 53.9, ease_out_cubic))
        mark = mark.with_opacity(lp)
        # breathing glow
        bw = 1 + 0.04 * math.sin((t - 53) * 2.2)
        ms = mark.scaled(int(mark.w * 0.92 * bw), int(mark.h * 0.92 * bw))
        mx = (D.W - ms.w) / 2
        my = 300
        over(img, glow(ms, 26, 0.35, D.TEAL), mx, my, 0.9, blend="add")
        over(img, ms, mx, my, 1.0)
    if wm > 0:
        word = text_sprite("视频扒谱工具", D.PAPER, "sans_bold", 54, tracking=10.0)
        over(img, wipe(word, wm).with_opacity(wm), (D.W - word.w) / 2, 596)
    if tg > 0:
        tag = text_sprite("B站教学视频  →  完整吉他谱  /  A4 PDF", D.GREY, "sans", 20, tracking=4.0)
        over(img, wipe(tag, tg).with_opacity(0.85 * tg), (D.W - tag.w) / 2, 676)
    # slogan
    caption(img, "一键扒谱 · 专治手截", 0, 748, t, 55.4, 58.6, D.TEAL, "serif", 34, 3.0, 600, rise=14, dur=0.6, align="center")
    # footer mono
    if t > 56.4:
        p = remap(t, 56.4, 57.2, ease_out_cubic)
        foot = text_sprite("BILIBILI  →  PDF   ·   LOCAL  ·  FREE", D.GREY_DIM, "sans", 13, tracking=4.0)

    # fade to black
    end = remap(t, 58.9, 60.0, ease_in_out_cubic)

        img = img * (1 - end)
